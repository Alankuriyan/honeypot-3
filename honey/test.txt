import subprocess
import json
import time
from pathlib import Path
from datetime import datetime


BASE_DIR = Path(__file__).parent
DECOY_FILE = str((BASE_DIR / "decoy.txt").resolve())
LOG_FILE = BASE_DIR / "access_log.txt"

last_time = None


def check_for_access():

    global last_time

    command = [
        "powershell",
        "-NoProfile",
        "-Command",
        """
        Get-WinEvent -FilterHashtable @{LogName='Security'; Id=4663} -MaxEvents 20 |
        Select-Object TimeCreated, Message |
        ConvertTo-Json -Compress
        """
    ]

    result = subprocess.run(
        command,
        capture_output=True,
        text=True
    )

    if result.returncode != 0:
        print("Could not read Security log")
        return

    if not result.stdout.strip():
        return

    try:
        events = json.loads(result.stdout)

        # If only one event is returned
        if isinstance(events, dict):
            events = [events]

        # Process oldest → newest
        events.reverse()

        for event in events:

            event_time = event["TimeCreated"]
            message = event["Message"]

            # Ignore events we've already processed
            if last_time is not None and event_time <= last_time:
                continue

            last_time = event_time

            # Is our decoy involved?
            if DECOY_FILE.lower() in message.lower():

                timestamp = datetime.now().strftime(
                    "%Y-%m-%d %H:%M:%S"
                )

                log_message = (
                    f"[{timestamp}] 🚨 DECOY ACCESSED\n"
                    f"File: {DECOY_FILE}\n"
                    f"Windows Event: 4663\n"
                    f"Event Time: {event_time}\n\n"
                )

                with open(
                    LOG_FILE,
                    "a",
                    encoding="utf-8"
                ) as file:

                    file.write(log_message)

                print(log_message)

    except Exception as error:

        print("Processing error:", error)


print("================================")
print("       HONEYPOT MONITOR")
print("================================")
print("Watching:", DECOY_FILE)
print("Waiting for access...")
print("Press CTRL+C to stop.\n")


while True:

    check_for_access()

    time.sleep(2)
